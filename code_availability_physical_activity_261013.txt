#1-1. Taxonomic profiling using MetaPhlAn4 [v.4.0.6]
#!/usr/bin/bash

INPUT_DIR="/path/to/QC_finished_cleaned_fastq_files"    
OUTPUT_DIR="/path/to/metaphlan_results"
DB_PATH="/path/to/metaphlan_db" # (e.g., mpa_vOct22_CHOCOPhlAnSGB_202212)
THREADS= "Threads"

mkdir -p "${OUTPUT_DIR}"

for fq in "${INPUT_DIR}"/*.fastq.gz; do
    [ -e "$fq" ] || continue

    # Extract sample name
    sample=$(basename "$fq" .fastq.gz)

    echo "Running MetaPhlAn4 for ${sample}..."

    metaphlan "$fq" \
        --input_type fastq \
        --nproc "${THREADS}" \
        --bowtie2db "${DB_PATH}" \
        --bowtie2out "${OUTPUT_DIR}/${sample}_bowtie2.txt" \
        -o "${OUTPUT_DIR}/${sample}_sgb_profile.txt"

    echo "Finished ${sample}"
done

#1-2.  Merging feature tables of all samples using HUMAnN3 [v.3.0]
merge_metaphlan_tables.py "/path/to/metaphlan_results"/*_sgb_profile.txt \
    > "/path/to/metaphlan_results"/merged_sgb_feature_table.txt

#1-3. Conversion SGB profiles into GTDB profiles using HUMAnN3 [v.3.0]

sgb_to_gtdb_profile.py \
    -i "/path/to/metaphlan_results"/merged_sgb_feature_table.txt \
    -d "/path/to/SGB2GTDB/indexing/file"   # (e.g.,mpa_vOct22_CHOCOPhlAnSGB_202212_SGB2GTDB.tsv) \
    -o "/path/to/metaphlan_results"/merged_gtdb_feature_table.txt

#2-1. Functional profiling using HUMAnN3 [v.3.0]
#!/usr/bin/bash

INPUT_DIR="/path/to/QC_finished_cleaned_fastq_files"
OUTPUT_DIR="/path/to/humann_results"
NUC_DB="/path/to/chocophlan_db"     # (e.g., chocophlan db v201901_v31)
PROT_DB="/path/to/uniref_db"        # (e.g., uniref90_201901b_full.dmnd)
THREADS="Threads"

mkdir -p "${OUTPUT_DIR}"

for fq in "${INPUT_DIR}"/*.fastq.gz; do
    [ -e "$fq" ] || continue

    # Extract sample name
    sample=$(basename "$fq" .fastq.gz)

    echo "Running HUMAnN for ${sample}..."

    humann \
        --input "${fq}" \
        --output "${OUTPUT_DIR}/${sample}_humann" \
        --threads "${THREADS}" \
        --nucleotide-database "${NUC_DB}" \
        --protein-database "${PROT_DB}"

    echo "Finished ${sample}"
done

#2-2. Normalization of metabolic pathway (MetaCyc) feature tables
#!/usr/bin/bash

INPUT_DIR="/path/to/humann_results"
OUTPUT_DIR="/path/to/humann_results_normalized"
mkdir -p "${OUTPUT_DIR}"

for tsv in "${INPUT_DIR}"/*_humann/*_pathabundance.tsv; do
    [ -e "$tsv" ] || continue

    sample=$(basename "$tsv" _pathabundance.tsv)

    echo "Normalizing HUMAnN pathabundance table for ${sample}..."

    humann_renorm_table \
        --input "${tsv}" \
        --output "${OUTPUT_DIR}/${sample}_pathabundance_relab.tsv" \
        --units relab

    echo "Finished ${sample}"
done

#2-3. Merging metabolic pathways feature tables of all samples using HUMAnN3 [v.3.0]
merge_metaphlan_tables.py "/path/to/humann_results_normalized"/*_pathabundance_relab.tsv \
    > "/path/to/humann_results_normalized"/merged_pwy_normalized_feature_table.txt

#2-4. Generating stratified relative abundance bar plots for visualization of species-level contributions to metabolic pathways using HUMAnN3 [v.3.0]
humann_barplot \
  --input "merged_pwy_normalized_feature_table_with_metadata" \
  --last-metadata "last_metadata_name" \ # e.g., sex
  --focal-feature "focal_feature" \ # e.g., PWY-5667
  --focal-metadata "focal_metadata" \ # e.g., group
  --sort sum metadata \
  --exclude-unclassified \
  --output "output" \
  --scaling logstack

#3-3. Correlation analysis for differential abundant bacterial species, MetaCyc metabolic pathways, GMMs, and predicted metabolites
# This step was performed in the R environment using the R package, Hmisc.
library(Hmisc)
feature_table <- read.delim("feature_table.txt", row.names = 1) # e.g., a concatenated feature matrix including relative abundance of differential abundant bacterial species, metabolic pathways and predicted metabolites
corr_output <- rcorr(as.matrix(feature_table), type = "spearman")
cormat <- as.matrix(corr_output$r)
pmat   <- as.matrix(corr_output$P)
upper_idx <- upper.tri(pmat, diag = FALSE)
pvals_vec <- pmat[upper_idx]
valid_idx <- !is.na(pvals_vec)
pvals_vec_valid <- pvals_vec[valid_idx]
qvals_vec_valid <- p.adjust(pvals_vec_valid, method = "BH")
qmat <- matrix(NA, nrow = nrow(pmat),  ncol = ncol(pmat), dimnames = dimnames(pmat))
qmat[upper_idx][valid_idx] <- qvals_vec_valid
qmat <- t(qmat)
qmat[upper_idx][valid_idx] <- qvals_vec_valid
diag(qmat) <- NA

#3-4. Mediation analysis
# This step was performed in the R environment using the R package, mediation.
library(mediation)
dat <- read.delim("metadata_with_clinical_phenotype_and_bacterial_taxa_feature_table.txt",check.names=FALSE")
mediator <-"taxon_name"
outcome <-"phenotype_name"
med_model <-lm(reformulate(c("age","sex","BMI","Group"),response=mediator),data=dat)
out_model <-lm(reformulate(c("age","sex","BMI","mediator"),response=outcome),data=dat)
med_result <-mediate(
    med_model,
    out_model,
    treat="Group",
    mediator=mediator,
    control.value="reference_group"
    treat.value="treat_group",
    boot=TRUE,
    sims=500)
summary(med_result)
